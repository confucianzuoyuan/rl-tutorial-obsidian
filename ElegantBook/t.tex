\documentclass[lang=cn,newtx,10pt,scheme=chinese]{elegantbook}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    width=12cm, height=6cm,
    xlabel={$x$},
    ylabel={密度},
    domain=-5:10,
    samples=200,
    legend pos=north west,
    grid=major
]

% 真实分布（双峰）
\addplot[blue, very thick] {0.4*exp(-((x-0)^2)/2)/sqrt(2*pi) + 
                             0.6*exp(-((x-5)^2)/1.5)/sqrt(2*pi*1.5)};
\addlegendentry{真实分布}

% 单个高斯（拟合差）
\addplot[red, dashed, thick] {exp(-((x-2.5)^2)/8)/sqrt(2*pi*8)};
\addlegendentry{单高斯（K=1）}

% 两个高斯混合（拟合好）
\addplot[green!60!black, thick] {0.4*exp(-((x-0)^2)/2)/sqrt(2*pi) + 
                                  0.6*exp(-((x-5)^2)/1.5)/sqrt(2*pi*1.5)};
\addlegendentry{GMM（K=2）}

\end{axis}
\end{tikzpicture}
\end{document}
